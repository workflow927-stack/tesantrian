function aksiUtamaCustomer() { const namaSales = getAkunAktif(); if(!namaSales) return; if (statusSayaGlobal === "Standby") { tampilkanModalKonfirmasi("Konfirmasi", "Sudah membawa customer?", function() { sedangProsesAksi = true; const btnCustomer = document.getElementById('btn-utama-customer'); if (btnCustomer) { btnCustomer.disabled = true; btnCustomer.className = "btn-primary btn-disabled"; btnCustomer.innerHTML = ` Memproses...`; } // Update UI lokal seketika jadi Melayani statusSayaGlobal = "Melayani"; renderUIRental(daftarAntrianGlobal, statusSistemGlobal, waktuStandbyHAGlobal, waktuStandbyAVGlobal); fetch(`${SCRIPT_URL}?aksi=ambilAntrian&namaSales=${encodeURIComponent(namaSales)}`, { method: 'GET', redirect: 'follow' }) .then(res => res.json()) .then(res => { sedangProsesAksi = false; if(res.status === "sukses") { muatDataAntrian(true); } else { tampilkanModal("Perhatian", res.pesan); muatDataAntrian(true); } }).catch(() => { sedangProsesAksi = false; muatDataAntrian(true); }); }); } } function aksiSelesaiClosing() { const namaSales = getAkunAktif(); if(!namaSales) return; tampilkanModalKonfirmasi("Konfirmasi Selesai", "Selesai melayani customer? Anda akan dikeluarkan dari antrian.", function() { sedangProsesAksi = true; const btnClosing = document.getElementById('btn-selesai-closing'); if (btnClosing) { btnClosing.disabled = true; btnClosing.className = "btn-danger btn-disabled"; btnClosing.innerHTML = ` Memproses...`; } // Hapus data secara lokal seketika daftarAntrianGlobal = daftarAntrianGlobal.filter(item => item.nama.trim().toLowerCase() !== namaSales.trim().toLowerCase()); statusSayaGlobal = "belum_ada"; renderUIRental(daftarAntrianGlobal, statusSistemGlobal, waktuStandbyHAGlobal, waktuStandbyAVGlobal); fetch(`${SCRIPT_URL}?aksi=selesaiMelayani&namaSales=${encodeURIComponent(namaSales)}`, { method: 'GET', redirect: 'follow' }) .then(res => res.json()) .then(res => { sedangProsesAksi = false; muatDataAntrian(true); }).catch(() => { sedangProsesAksi = false; muatDataAntrian(true); }); }); }
